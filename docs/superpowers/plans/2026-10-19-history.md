# Lịch sử thay đổi — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ghi nhật ký mọi thay đổi trên nhiệm vụ, sự kiện, tag và dòng ghi chú (giữ 30 ngày), có trang "Lịch sử" để xem lại và khôi phục mục đã xóa.

**Architecture:** Mỗi hàm ghi trong `src/lib/firebase/*` ghi thêm một tài liệu `users/{uid}/history/{id}` **trong cùng batch/transaction**. Trang Lịch sử đọc bằng `getDocs` theo trang, không dùng realtime listener.

**Tech Stack:** Next.js 16 (App Router, client components), React 19, antd 6, dayjs, Firebase JS SDK 12 (Firestore), Vitest 4, `@firebase/rules-unit-testing` chạy với Firestore emulator.

```ts
import "dayjs/locale/vi";
import dayjs, { type Dayjs } from "dayjs";
import { reminderOptions } from "@/lib/planner";

export const historyActions = [
  "create",
  "update",
  "delete",
  "complete",
  "reopen",
  "convert",
  "restore",
] as const;
export const historyEntities = ["task", "event", "tag", "noteItem"] as const;

export type HistoryAction = (typeof historyActions)[number];
export type HistoryEntity = (typeof historyEntities)[number];
/** A document's fields as they were written, with Timestamps read back as Dates. */
export type HistorySnapshot = Record<string, unknown>;

export interface HistoryInput {
  action: HistoryAction;
  entity: HistoryEntity;
  entityId: string;
  title: string;
  before: HistorySnapshot | null;
  after: HistorySnapshot | null;
  /** How many items one save touched; 1 unless a batch created several. */
  count?: number;
}

export interface HistoryEntry extends Omit<HistoryInput, "count"> {
  id: string;
  count: number;
  at: Date;
}

export type HistoryFilter = "all" | HistoryEntity;

export const historyRetentionDays = 30;
const dayMs = 86_400_000;
const titleLimit = 200;

/** The oldest moment the history view still shows. */
export function historyCutoff(now: Date) {
  return new Date(now.getTime() - historyRetentionDays * dayMs);
}

/** When a new entry becomes eligible for Firestore's TTL cleanup. */
export function historyExpiry(now: Date) {
  return new Date(now.getTime() + historyRetentionDays * dayMs);
}

/** Fits a title or checklist line into the rules' 200-character limit. */
export function historyTitle(text: string) {
  return text.trim().slice(0, titleLimit);
}

const none = "Không có";
const statusLabels: Record<string, string> = {
  todo: "Chưa xong",
  in_progress: "Đang làm",
  done: "Đã xong",
};
const priorityLabels: Record<string, string> = {
  low: "Thấp",
  medium: "Trung bình",
  high: "Cao",
};

function formatText(limit?: number) {
  return (value: unknown) => {
    const text = typeof value === "string" ? value : "";
    if (!text) return none;
    return limit && text.length > limit ? `${text.slice(0, limit)}…` : text;
  };
}

function formatDate(value: unknown) {
  return value instanceof Date ? dayjs(value).format("DD/MM/YYYY") : none;
}

function formatReminder(value: unknown) {
  const minutes = typeof value === "number" ? value : -1;
  return reminderOptions.find((option) => option.value === minutes)?.label ?? `${minutes} phút`;
}

function formatLabel(labels: Record<string, string>) {
  return (value: unknown) => labels[String(value)] ?? String(value ?? none);
}

// Order here is the order changes are listed in. `deadline` is folded into
// `startAt` by `normalize`, so a task and an event line up when converted.
const shownFields: Array<{ field: string; label: string; format: (value: unknown) => string }> = [
  { field: "title", label: "Tiêu đề", format: formatText() },
  { field: "name", label: "Tên tag", format: formatText() },
  { field: "text", label: "Nội dung", format: formatText() },
  { field: "description", label: "Ghi chú", format: formatText(80) },
  { field: "location", label: "Địa điểm", format: formatText() },
  { field: "startAt", label: "Bắt đầu", format: formatDate },
  { field: "endAt", label: "Kết thúc", format: formatDate },
  { field: "status", label: "Trạng thái", format: formatLabel(statusLabels) },
  { field: "priority", label: "Ưu tiên", format: formatLabel(priorityLabels) },
  { field: "tagName", label: "Tag", format: formatText() },
  { field: "tagColor", label: "Màu", format: formatText() },
  { field: "color", label: "Màu", format: formatText() },
  { field: "reminderMinutes", label: "Nhắc nhở", format: formatReminder },
];

/** Fields whose values are hex colors, so the view can draw a swatch. */
export const colorFields: ReadonlySet<string> = new Set(["tagColor", "color"]);

export interface FieldChange {
  field: string;
  label: string;
  from: string;
  to: string;
}

function normalize(snapshot: HistorySnapshot): HistorySnapshot {
  if (!("deadline" in snapshot)) return snapshot;
  const { deadline, ...rest } = snapshot;
  return { ...rest, startAt: deadline };
}

function sameValue(a: unknown, b: unknown) {
  if (a instanceof Date && b instanceof Date) return a.getTime() === b.getTime();
  return a === b;
}

/**
 * The shown fields that differ between two snapshots, formatted for reading.
 * A field present on only one side (e.g. a task's priority after converting
 * an event) is skipped rather than shown as appearing from nowhere.
 */
export function diffFields(
  before: HistorySnapshot | null,
  after: HistorySnapshot | null,
): FieldChange[] {
  if (!before || !after) return [];
  const from = normalize(before);
  const to = normalize(after);
  return shownFields.flatMap(({ field, label, format }) =>
    field in from && field in to && !sameValue(from[field], to[field])
      ? [{ field, label, from: format(from[field]), to: format(to[field]) }]
      : [],
  );
}

const nouns: Record<HistoryEntity, string> = {
  task: "nhiệm vụ",
  event: "sự kiện",
  tag: "tag",
  noteItem: "dòng ghi chú",
};

/** One sentence saying what happened, e.g. `Đã xóa nhiệm vụ “Nộp báo cáo”`. */
export function describeEntry(
  entry: Pick<HistoryEntry, "action" | "entity" | "title" | "count">,
): string {
  const noun = nouns[entry.entity];
  const quoted = `“${entry.title}”`;
  const many = entry.count > 1 ? `${entry.count} ` : "";
  switch (entry.action) {
    case "create":
      return `${entry.entity === "noteItem" ? "Đã thêm" : "Đã tạo"} ${many}${noun} ${quoted}`;
    case "update":
      return `Đã sửa ${noun} ${quoted}`;
    case "delete":
      // Ticking a checklist line off is how it gets deleted.
      return `${entry.entity === "noteItem" ? "Đã tích" : "Đã xóa"} ${noun} ${quoted}`;
    case "complete":
      return `Đã hoàn thành ${noun} ${quoted}`;
    case "reopen":
      return `Đã mở lại ${noun} ${quoted}`;
    case "convert": {
      const from = entry.entity === "task" ? nouns.event : nouns.task;
      return `Đã đổi ${from} ${quoted} thành ${many}${noun}`;
    }
    case "restore":
      return `Đã khôi phục ${noun} ${quoted}`;
  }
}

export interface HistoryDay {
  key: string;
  label: string;
  entries: HistoryEntry[];
}

function dayLabel(day: Dayjs, today: Dayjs) {
  if (day.isSame(today, "day")) return "Hôm nay";
  if (day.isSame(today.subtract(1, "day"), "day")) return "Hôm qua";
  const label = day.locale("vi").format("dddd, DD/MM");
  return label.charAt(0).toUpperCase() + label.slice(1);
}

/** Groups newest-first entries by local day, keeping their order. */
export function groupByDay(entries: HistoryEntry[], today: Dayjs): HistoryDay[] {
  const days: HistoryDay[] = [];
  for (const entry of entries) {
    const day = dayjs(entry.at);
    const key = day.format("YYYY-MM-DD");
    const last = days.at(-1);
    if (last?.key === key) {
      last.entries.push(entry);
    } else {
      days.push({ key, label: dayLabel(day, today), entries: [entry] });
    }
  }
  return days;
}

export function filterEntries(
  entries: HistoryEntry[],
  filter: HistoryFilter,
  deletedOnly: boolean,
) {
  return entries.filter(
    (entry) =>
      (filter === "all" || entry.entity === filter) &&
      (!deletedOnly || entry.action === "delete"),
  );
}

/** Identifies an item across kinds, for the set of items currently loaded. */
export function historyItemKey(entity: HistoryEntity, id: string) {
  return `${entity}:${id}`;
}

/** A deleted item can come back while it is still gone from the loaded data. */
export function isRestorable(entry: HistoryEntry, existing: ReadonlySet<string>) {
  return (
    entry.action === "delete" &&
    entry.before !== null &&
    !existing.has(historyItemKey(entry.entity, entry.entityId))
  );
}
```

```ts
  historyTitle,
  type HistoryAction,
  type HistoryEntity,
  type HistoryEntry,
  type HistoryInput,
  type HistorySnapshot,

export const historyPageSize = 50;

function historyCollection(userId: string) {
  return collection(getFirestoreDb(), "users", userId, "history");
}

function plainValue(value: unknown): unknown {
  if (value instanceof Timestamp) return value.toDate();
  if (Array.isArray(value)) return value.map(plainValue);
  if (value && typeof value === "object" && !(value instanceof Date)) {
    return Object.fromEntries(
      Object.entries(value).map(([key, entry]) => [key, plainValue(entry)]),
    );
  }
  return value;
}

/**
 * A stored document's fields without its bookkeeping timestamps, with nested
 * Timestamps turned into Dates so snapshots compare and format as plain values.
 */
export function toSnapshot(data: DocumentData): HistorySnapshot {
  return Object.fromEntries(
    Object.entries(data)
      .filter(([key]) => key !== "createdAt" && key !== "updatedAt")
      .map(([key, value]) => [key, plainValue(value)]),
  );
}

/** True when two payloads differ in a field the history view shows. */
export function payloadChanged(before: DocumentData, after: DocumentData) {
  return diffFields(toSnapshot(before), toSnapshot(after)).length > 0;
}

/**
 * The entry to write alongside a change. Callers `set` it in the same batch or
 * transaction as the change itself, so both land or neither does.
 */
export function historyWrite(
  userId: string,
  input: HistoryInput,
): { ref: DocumentReference; data: DocumentData } {
  return {
    ref: doc(historyCollection(userId)),
    data: {
      action: input.action,
      entity: input.entity,
      entityId: input.entityId,
      title: historyTitle(input.title),
      before: input.before,
      after: input.after,
      count: input.count ?? 1,
      at: serverTimestamp(),
      expiresAt: Timestamp.fromDate(historyExpiry(new Date())),
    },
  };
}

function historyFromDocument(snapshot: QueryDocumentSnapshot<DocumentData>): HistoryEntry {
  const data = snapshot.data();
  return {
    id: snapshot.id,
    action: data.action as HistoryAction,
    entity: data.entity as HistoryEntity,
    entityId: String(data.entityId ?? ""),
    title: String(data.title ?? ""),
    before: data.before ? toSnapshot(data.before) : null,
    after: data.after ? toSnapshot(data.after) : null,
    count: typeof data.count === "number" ? data.count : 1,
    at: readDate(data.at) ?? new Date(),
  };
}

export interface HistoryPage {
  entries: HistoryEntry[];
  /** Where the next page starts, or null once everything is loaded. */
  cursor: QueryDocumentSnapshot<DocumentData> | null;
}

/**
 * One page of entries, newest first. TTL can lag by about a day, so entries
 * past the 30-day window are filtered out here rather than trusted to be gone.
 */
export async function fetchHistoryPage(
  userId: string,
  after: QueryDocumentSnapshot<DocumentData> | null = null,
): Promise<HistoryPage> {
  const constraints: QueryConstraint[] = [
    where("at", ">=", Timestamp.fromDate(historyCutoff(new Date()))),
    orderBy("at", "desc"),
    ...(after ? [startAfter(after)] : []),
    limit(historyPageSize),
  ];
  const snapshot = await getDocs(query(historyCollection(userId), ...constraints));
  return {
    entries: snapshot.docs.map(historyFromDocument),
    cursor: snapshot.docs.length === historyPageSize ? snapshot.docs.at(-1) ?? null : null,
  };
}

/** Deletes every entry, 100 at a time (the rules' list limit). */
export async function clearHistory(userId: string) {
  const nextBatch = () => getDocs(query(historyCollection(userId), limit(100)));
  let snapshot = await nextBatch();
  while (!snapshot.empty) {
    const batch = writeBatch(getFirestoreDb());
    snapshot.docs.forEach((item) => batch.delete(item.ref));
    await batch.commit();
    snapshot = await nextBatch();
  }
}
```

```ts
import {
  doc,
  runTransaction,
  serverTimestamp,
  writeBatch,
  type CollectionReference,
  type DocumentData,
  type DocumentReference,
  type Transaction,
  type WriteBatch,
} from "firebase/firestore";
import { getFirestoreDb } from "@/lib/firebase/client";
import { historyWrite, payloadChanged, toSnapshot } from "@/lib/firebase/history";
import type { HistoryEntity } from "@/lib/history";

/** How one collection's documents are read, titled and logged. */
export interface LoggedKind<P extends DocumentData> {
  entity: HistoryEntity;
  collection: CollectionReference;
  /** Reads a stored document into the shape the writer stores, normalizing older ones. */
  readPayload: (data: DocumentData & { id: string }) => P;
  titleOf: (payload: P) => string;
  /** Shown when the document to change or delete is already gone. */
  missingMessage: string;
}

type Write =
  | { op: "set"; ref: DocumentReference; data: DocumentData }
  | { op: "update"; ref: DocumentReference; data: DocumentData }
  | { op: "delete"; ref: DocumentReference };

function applyToBatch(batch: WriteBatch, writes: Write[]) {
  for (const write of writes) {
    if (write.op === "set") batch.set(write.ref, write.data);
    else if (write.op === "update") batch.update(write.ref, write.data);
    else batch.delete(write.ref);
  }
}

function applyToTransaction(transaction: Transaction, writes: Write[]) {
  for (const write of writes) {
    if (write.op === "set") transaction.set(write.ref, write.data);
    else if (write.op === "update") transaction.update(write.ref, write.data);
    else transaction.delete(write.ref);
  }
}

async function readPayload<P extends DocumentData>(
  transaction: Transaction,
  kind: LoggedKind<P>,
  ref: DocumentReference,
) {
  const snapshot = await transaction.get(ref);
  if (!snapshot.exists()) throw new Error(kind.missingMessage);
  return kind.readPayload({ ...snapshot.data(), id: snapshot.id });
}

/**
 * Saves one document per payload, logged in the same commit. The first
 * payload updates `existingId` when given; the rest are created and logged as
 * one entry. `replaces` (a plan of the other kind) is deleted and the save is
 * logged as a conversion. Plain creates use a batch so they still queue
 * offline; anything that must read the old document uses a transaction.
 * Resolves with the created documents, in payload order.
 */
export async function saveLoggedSeries<P extends DocumentData>(
  userId: string,
  kind: LoggedKind<P>,
  existingId: string | null,
  payloads: P[],
  replaces?: DocumentReference,
): Promise<DocumentReference[]> {
  const existingRef = existingId ? doc(kind.collection, existingId) : null;
  const createdPayloads = existingRef ? payloads.slice(1) : payloads;
  const createdRefs = createdPayloads.map(() => doc(kind.collection));

  const writes = (before: P | null, replaced: DocumentData | null): Write[] => {
    const list: Write[] = [];
    if (replaces) list.push({ op: "delete", ref: replaces });
    if (existingRef && before && payloads[0]) {
      const after = payloads[0];
      list.push({ op: "update", ref: existingRef, data: { ...after, updatedAt: serverTimestamp() } });
      if (payloadChanged(before, after)) {
        list.push({
          op: "set",
          ...historyWrite(userId, {
            action: "update",
            entity: kind.entity,
            entityId: existingRef.id,
            title: kind.titleOf(after),
            before,
            after,
          }),
        });
      }
    }
    createdRefs.forEach((ref, index) =>
      list.push({
        op: "set",
        ref,
        data: { ...createdPayloads[index], createdAt: serverTimestamp(), updatedAt: serverTimestamp() },
      }),
    );
    if (createdRefs.length) {
      list.push({
        op: "set",
        ...historyWrite(userId, {
          action: replaces ? "convert" : "create",
          entity: kind.entity,
          entityId: createdRefs[0].id,
          title: kind.titleOf(createdPayloads[0]),
          before: replaced ? toSnapshot(replaced) : null,
          after: createdPayloads[0],
          count: createdRefs.length,
        }),
      });
    }
    return list;
  };

  if (!existingRef && !replaces) {
    const batch = writeBatch(getFirestoreDb());
    applyToBatch(batch, writes(null, null));
    await batch.commit();
    return createdRefs;
  }

  await runTransaction(getFirestoreDb(), async (transaction) => {
    const before = existingRef ? await readPayload(transaction, kind, existingRef) : null;
    const replaced = replaces ? await transaction.get(replaces) : null;
    applyToTransaction(transaction, writes(before, replaced?.exists() ? replaced.data() : null));
  });
  return createdRefs;
}

/** Deletes a document and logs its last contents, so it can be restored. */
export function deleteLogged<P extends DocumentData>(
  userId: string,
  kind: LoggedKind<P>,
  id: string,
) {
  const ref = doc(kind.collection, id);
  return runTransaction(getFirestoreDb(), async (transaction) => {
    const before = await readPayload(transaction, kind, ref);
    transaction.delete(ref);
    const entry = historyWrite(userId, {
      action: "delete",
      entity: kind.entity,
      entityId: id,
      title: kind.titleOf(before),
      before,
      after: null,
    });
    transaction.set(entry.ref, entry.data);
  });
}
```

```ts
/**
 * Reads the stored list, applies `update` and writes it back in one
 * transaction, so concurrent edits from other devices are never lost. Creates
 * the document on first use (title/color/pinned are fixed by the rules' shape).
 * A `history` entry returned by `update` is logged in the same transaction.
 */
function updateChecklist<T>(
  userId: string,
  update: (
    items: NoteItem[],
  ) => { items: NoteItem[]; result: T; history?: HistoryInput } | null,
) {
  const ref = checklistRef(userId);
  return runTransaction(getFirestoreDb(), async (transaction: Transaction) => {
    const snapshot = await transaction.get(ref);
    const current = snapshot.exists() ? readItems(snapshot.data().items) : [];
    const next = update(current);
    if (!next) return null;
    const items = cleanItems(next.items);
    if (snapshot.exists()) {
      transaction.update(ref, { items, updatedAt: serverTimestamp() });
    } else {
      transaction.set(ref, {
        title: "",
        color: "",
        pinned: false,
        items,
        createdAt: serverTimestamp(),
        updatedAt: serverTimestamp(),
      });
    }
    if (next.history) {
      const entry = historyWrite(userId, next.history);
      transaction.set(entry.ref, entry.data);
    }
    return next.result;
  });
}
```

```ts
export function addChecklistItems(userId: string, newItems: NoteItem[]) {
  return updateChecklist(userId, (items) => {

    const added = cleanItems(newItems).slice(0, room);

    return {

        entity: "noteItem",
        entityId: added[0].id,

/** Edits a line's text. Clearing it deletes the line, logged as a delete. */
export function updateChecklistItem(userId: string, itemId: string, text: string) {
  return updateChecklist(userId, (items) => {
    const index = items.findIndex((item) => item.id === itemId);
    if (index === -1) return null;
    const old = items[index];
    const nextText = text.trim().slice(0, noteLimits.itemText);
    if (nextText === old.text) return null;
    const history: HistoryInput = nextText
      ? {
          action: "update",
          entity: "noteItem",
          entityId: itemId,
          title: nextText,
          before: { id: itemId, text: old.text },
          after: { id: itemId, text: nextText },
        }
      : {
          action: "delete",
          entity: "noteItem",
          entityId: itemId,
          title: old.text,
          before: { id: itemId, text: old.text, index },
          after: null,
        };
    return {
      items: items.map((item) => (item.id === itemId ? { ...item, text } : item)),
      result: true,
      history,
    };
  });
}

/** Ticks a line off by deleting it. Resolves with it and its position, for undo. */
export function removeChecklistItem(userId: string, itemId: string) {
  return updateChecklist(userId, (items) => {
    const index = items.findIndex((item) => item.id === itemId);
    if (index === -1) return null;
    const item = items[index];
    return {
      items: items.filter((entry) => entry.id !== itemId),
      result: { item, index },
      history: {
        action: "delete",
        entity: "noteItem",
        entityId: itemId,
        title: item.text,
        before: { id: item.id, text: item.text, index },
        after: null,
      },
    };
  });
}

/** Undo for `removeChecklistItem`: puts the line back where it was. */
export function restoreChecklistItem(userId: string, item: NoteItem, index: number) {
  return updateChecklist(userId, (items) => {
    if (items.some((entry) => entry.id === item.id)) return null;
    const next = [...items];
    next.splice(Math.min(index, next.length), 0, item);
    return {
      items: next,
      result: true,
      history: {
        action: "restore",
        entity: "noteItem",
        entityId: item.id,
        title: item.text,
        before: null,
        after: { id: item.id, text: item.text },
      },
    };
  });
}
```

```ts
import { doc, runTransaction, serverTimestamp } from "firebase/firestore";
import { getFirestoreDb } from "@/lib/firebase/client";
import { historyWrite } from "@/lib/firebase/history";
import { restoreChecklistItem } from "@/lib/firebase/notes";
import type { HistoryEntry } from "@/lib/history";

const collections = { task: "tasks", event: "events", tag: "tags" } as const;
const alreadyRestored = "Mục này đã được khôi phục.";

/**
 * Puts a deleted item back under its old id, so a tag reconnects with the
 * plans still pointing at it. The rules require a fresh `createdAt`, so the
 * restored item counts as new in creation order.
 */
export async function restoreFromHistory(userId: string, entry: HistoryEntry) {
  const before = entry.before;
  if (entry.action !== "delete" || !before) {
    throw new Error("Mục này không thể khôi phục.");
  }

  if (entry.entity === "noteItem") {
    // Logs its own "restore" entry, like the undo on the notes page.
    const restored = await restoreChecklistItem(
      userId,
      { id: entry.entityId, text: String(before.text ?? entry.title) },
      typeof before.index === "number" ? before.index : 0,
    );
    if (!restored) throw new Error(alreadyRestored);
    return;
  }

  const db = getFirestoreDb();
  const ref = doc(db, "users", userId, collections[entry.entity], entry.entityId);
  await runTransaction(db, async (transaction) => {
    if ((await transaction.get(ref)).exists()) throw new Error(alreadyRestored);
    transaction.set(ref, { ...before, createdAt: serverTimestamp(), updatedAt: serverTimestamp() });
    const log = historyWrite(userId, {
      action: "restore",
      entity: entry.entity,
      entityId: entry.entityId,
      title: entry.title,
      before: null,
      after: before,
    });
    transaction.set(log.ref, log.data);
  });
}
```

```ts
"use client";

import type { DocumentData, QueryDocumentSnapshot } from "firebase/firestore";
import { useEffect, useState } from "react";
import { clearHistory, fetchHistoryPage } from "@/lib/firebase/history";
import { restoreFromHistory } from "@/lib/firebase/restore";
import type { HistoryEntry } from "@/lib/history";

function asError(error: unknown) {
  return error instanceof Error ? error : new Error("Không thể tải lịch sử.");
}

/**
 * Loads the history log a page at a time. It is fetched on open and after each
 * restore or clear rather than kept live, so no listener runs in the background.
 */
export function useHistory(userId: string) {
  const [entries, setEntries] = useState<HistoryEntry[]>([]);
  const [cursor, setCursor] = useState<QueryDocumentSnapshot<DocumentData> | null>(null);
  const [loading, setLoading] = useState(true);
  const [loadingMore, setLoadingMore] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const [version, setVersion] = useState(0);

  useEffect(() => {
    let cancelled = false;
    fetchHistoryPage(userId).then(
      (page) => {
        if (cancelled) return;
        setEntries(page.entries);
        setCursor(page.cursor);
        setError(null);
        setLoading(false);
      },
      (nextError: unknown) => {
        if (cancelled) return;
        setError(asError(nextError));
        setLoading(false);
      },
    );
    return () => {
      cancelled = true;
    };
  }, [userId, version]);

  const reload = () => {
    setLoading(true);
    setVersion((value) => value + 1);
  };

  const loadMore = async () => {
    if (!cursor || loadingMore) return;
    setLoadingMore(true);
    try {
      const page = await fetchHistoryPage(userId, cursor);
      setEntries((current) => [...current, ...page.entries]);
      setCursor(page.cursor);
    } catch (nextError) {
      setError(asError(nextError));
    } finally {
      setLoadingMore(false);
    }
  };

  /** Throws on failure so the page can say why. */
  const restore = async (entry: HistoryEntry) => {
    await restoreFromHistory(userId, entry);
    reload();
  };

  const clear = async () => {
    await clearHistory(userId);
    reload();
  };

  return { entries, loading, loadingMore, error, hasMore: cursor !== null, loadMore, reload, restore, clear };
}
```

```tsx
interface HistoryPageProps {
  userId: string;
  /** `historyItemKey`s of every item loaded now, to tell which deleted ones are still gone. */
  existingKeys: ReadonlySet<string>;
}

export function HistoryPage({ userId, existingKeys }: HistoryPageProps) {
  const { message, modal } = App.useApp();
  const history = useHistory(userId);
  const [filter, setFilter] = useState<HistoryFilter>("all");
  const [deletedOnly, setDeletedOnly] = useState(false);
  const [restoringId, setRestoringId] = useState<string | null>(null);

  const days = useMemo(
    () => groupByDay(filterEntries(history.entries, filter, deletedOnly), dayjs()),
    [history.entries, filter, deletedOnly],
  );

  const restore = async (entry: HistoryEntry) => {
    setRestoringId(entry.id);
    try {
      await history.restore(entry);
      void message.success(`Đã khôi phục “${entry.title}”.`);
    } catch (error) {
      void message.error(firebaseErrorMessage(error, "Không thể khôi phục."));
    } finally {
      setRestoringId(null);
    }
  };

  const confirmClear = () =>
    modal.confirm({
      title: "Xóa toàn bộ nhật ký?",
      content: "Mục đã xóa sẽ không còn khôi phục được từ trang này.",
      okText: "Xóa nhật ký",
      okButtonProps: { danger: true },
      cancelText: "Hủy",
      onOk: async () => {
        try {
          await history.clear();
          void message.success("Đã xóa nhật ký.");
        } catch (error) {
          void message.error(firebaseErrorMessage(error, "Không thể xóa nhật ký."));
        }
      },
    });

  return (
    <section className="history-page" aria-label="Lịch sử">
      <div className="history-heading">
        <div className="notes-heading">
          <span className="planner-eyebrow">LỊCH SỬ</span>
          <h1>Nhật ký thay đổi</h1>
          <p>Mọi thay đổi trong {historyRetentionDays} ngày qua. Mục đã xóa có thể khôi phục.</p>
        </div>
        <div className="history-actions">
          <Button
            icon={<ReloadOutlined />}
            aria-label="Làm mới"
            loading={history.loading}
            onClick={history.reload}
          />
          <Dropdown
            trigger={["click"]}
            menu={{
              items: [
                { key: "clear", danger: true, label: "Xóa nhật ký", disabled: !history.entries.length },
              ],
              onClick: confirmClear,
            }}
          >
            <Button icon={<EllipsisOutlined />} aria-label="Thao tác khác" />
          </Dropdown>
        </div>
      </div>

      <div className="history-filters">
        <Segmented
          options={filterOptions}
          value={filter}
          onChange={(value) => setFilter(value as HistoryFilter)}
        />
        <label className="history-deleted-only">
          <Switch size="small" checked={deletedOnly} onChange={(checked) => setDeletedOnly(checked)} />
          Chỉ mục đã xóa
        </label>
      </div>

      {history.error ? (
        <Alert
          type="error"
          showIcon
          title="Không thể tải lịch sử"
          description={firebaseErrorMessage(history.error, "Vui lòng thử lại.")}
        />
      ) : null}

      {history.loading && !history.entries.length ? (
        <Skeleton active paragraph={{ rows: 4 }} />
      ) : days.length ? (
        days.map((day) => (
          <section key={day.key} className="history-day" aria-label={day.label}>
            <h2>{day.label}</h2>
            <ul>
              {day.entries.map((entry) => (
                <HistoryEntryRow
                  key={entry.id}
                  entry={entry}
                  restorable={isRestorable(entry, existingKeys)}
                  restoring={restoringId === entry.id}
                  onRestore={(item) => void restore(item)}
                />
              ))}
            </ul>
          </section>
        ))
      ) : (
        <EmptyState
          art={history.entries.length ? "search" : "tea"}
          title={
            history.entries.length
              ? "Không có thay đổi nào khớp bộ lọc."
              : `Chưa có thay đổi nào trong ${historyRetentionDays} ngày qua.`
          }
        />
      )}

      {history.hasMore ? (
        <Button className="history-more" loading={history.loadingMore} onClick={() => void history.loadMore()}>
          Tải thêm
        </Button>
      ) : null}
    </section>
  );
}
```
